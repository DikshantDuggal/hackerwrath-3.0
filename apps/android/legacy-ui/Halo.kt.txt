/**
 * Halo: the one living light that watches calls with the user. It breathes while listening,
 * warms to amber while checking, turns red for high risk, breaks into dashes when protection is
 * unavailable, and goes dim when scam check is off. It is decoration for sighted users; every
 * screen also says the same thing in words.
 */
enum class HaloMood { Day, Listening, Checking, Possible, High, Unavailable, Off }

private fun HaloMood.color(): Color = when (this) {
    HaloMood.Day -> HaloDay
    HaloMood.Listening -> HaloListening
    HaloMood.Checking, HaloMood.Possible -> HaloChecking
    HaloMood.High -> HaloHigh
    HaloMood.Unavailable -> HaloPaused
    HaloMood.Off -> HaloOff
}

private fun HaloMood.glow(): Float = when (this) {
    HaloMood.Day -> 0.45f
    HaloMood.Listening -> 0.6f
    HaloMood.Checking, HaloMood.Possible -> 0.9f
    HaloMood.High -> 1f
    HaloMood.Unavailable -> 0.2f
    HaloMood.Off -> 0f
}

private const val RIPPLE_SECONDS = 1.6f
private const val VOICE_BARS = 72

/**
 * @param voice draws the caller's voice as bars around the ring (in call).
 * @param ringing sends a ripple out every 1.4 s (incoming call).
 * @param pulseKey change it to send out three ripples once (a chime moment).
 * @param voiceLevel how loud the voice bars are; lower it while muted.
 */
@Composable
fun Halo(
    mood: HaloMood,
    modifier: Modifier = Modifier,
    voice: Boolean = false,
    ringing: Boolean = false,
    pulseKey: Any? = null,
    voiceLevel: Float = 1f,
) {
    val reduceMotion = LocalReduceMotion.current
    val color by animateColorAsState(mood.color(), tween(if (reduceMotion) 0 else 700), label = "haloColor")
    val glow by animateFloatAsState(mood.glow(), tween(if (reduceMotion) 0 else 700), label = "haloGlow")
    val dash by animateFloatAsState(if (mood == HaloMood.Unavailable) 1f else 0f, tween(if (reduceMotion) 0 else 500), label = "haloDash")
    val voiceAmount by animateFloatAsState(if (voice) voiceLevel else 0f, tween(if (reduceMotion) 0 else 500), label = "haloVoice")

    // A clock read only while drawing, so time passing redraws the Halo without recomposing it.
    var time by remember { mutableFloatStateOf(0f) }
    val ripples = remember { mutableStateListOf<Float>() }
    LaunchedEffect(reduceMotion) {
        if (reduceMotion) return@LaunchedEffect
        val start = withFrameNanos { it }
        while (true) {
            withFrameNanos { now ->
                time = (now - start) / 1_000_000_000f
                if (ripples.isNotEmpty() && time - ripples.first() > RIPPLE_SECONDS) ripples.removeAt(0)
            }
        }
    }
    LaunchedEffect(pulseKey, reduceMotion) {
        if (pulseKey == null || reduceMotion) return@LaunchedEffect
        repeat(3) {
            ripples.add(time)
            delay(220)
        }
    }
    LaunchedEffect(ringing, reduceMotion) {
        while (ringing && !reduceMotion) {
            ripples.add(time)
            delay(1_400)
        }
    }

    Canvas(modifier.clearAndSetSemantics { }) {
        val t = time
        val r = size.minDimension / 2f / 1.45f
        val c = center
        val fast = mood == HaloMood.Checking
        val breathe = 1f + if (fast) 0.05f * sin(t * 6.2f) else 0.035f * sin(t * 1.25f)

        if (glow > 0.01f) {
            drawCircle(
                brush = Brush.radialGradient(
                    0f to color.copy(alpha = 0.55f * glow),
                    0.45f to color.copy(alpha = 0.16f * glow),
                    1f to color.copy(alpha = 0f),
                    center = c,
                    radius = r * 2.2f,
                ),
                radius = r * 2.2f,
                center = c,
            )
        }

        for (startedAt in ripples) {
            val q = (t - startedAt) / RIPPLE_SECONDS
            if (q !in 0f..1f) continue
            drawCircle(
                color = color.copy(alpha = 0.6f * (1f - q)),
                radius = r * (0.8f + 1.5f * q),
                center = c,
                style = Stroke(width = (3f * (1f - q) + 0.5f) * density),
            )
        }

        if (voiceAmount > 0.02f) drawVoice(c, r, t, color, voiceAmount, reduceMotion)

        drawRing(c, r * breathe, 1.0f, t * 0.18f * (if (fast) 3f else 1f), color.copy(alpha = 0.55f), 1.8f * density, dash, t, 0)
        drawRing(c, r * breathe, 0.86f, -t * 0.25f * (if (fast) 3f else 1f), color.copy(alpha = 0.35f), 1.2f * density, dash, t, 1)

        val dim = if (glow < 0.05f) 0.55f else 1f
        val core = blob(c, r * 0.62f * breathe, t, 1f - dash)
        drawPath(
            core,
            brush = Brush.radialGradient(
                0f to lerp(color, Color.White, 0.55f).copy(alpha = 0.95f * dim),
                0.55f to color.copy(alpha = 0.95f * dim),
                1f to lerp(color, Color.Black, 0.3f).copy(alpha = 0.95f * dim),
                center = Offset(c.x - r * 0.18f, c.y - r * 0.22f),
                radius = r * 0.7f,
            ),
        )
    }
}

private fun DrawScope.drawVoice(c: Offset, r: Float, t: Float, color: Color, amount: Float, still: Boolean) {
    val r0 = r * 1.16f
    for (i in 0 until VOICE_BARS) {
        val amp = if (still) 0.35f else 0.25f + 0.75f * abs(sin(t * 2.1f + i * 0.37f) * sin(t * 0.9f + i * 0.13f))
        val a = (i.toFloat() / VOICE_BARS) * 2f * PI.toFloat() - PI.toFloat() / 2f
        val r1 = r0 + r * 0.22f * amp * amount
        drawLine(
            color = color.copy(alpha = 0.55f * amount.coerceAtMost(1f)),
            start = Offset(c.x + cos(a) * r0, c.y + sin(a) * r0),
            end = Offset(c.x + cos(a) * r1, c.y + sin(a) * r1),
            strokeWidth = 2.6f * density,
            cap = StrokeCap.Round,
        )
    }
}

private fun DrawScope.drawRing(
    c: Offset, r: Float, scale: Float, turns: Float, color: Color, width: Float, dash: Float, t: Float, index: Int,
) {
    val path = Path()
    val steps = 120
    for (i in 0..steps) {
        val a = i.toFloat() / steps * 2f * PI.toFloat()
        val wobble = 1f + 0.018f * sin(a * 3f + t * (1.1f + index)) * (1f - dash)
        val rr = r * scale * wobble
        val x = c.x + cos(a) * rr
        val y = c.y + sin(a) * rr
        if (i == 0) path.moveTo(x, y) else path.lineTo(x, y)
    }
    path.close()
    val effect = if (dash > 0.5f) PathEffect.dashPathEffect(floatArrayOf(10f * density, 9f * density)) else null
    rotate(degrees = turns * 57.2958f, pivot = c) {
        drawPath(path, color, style = Stroke(width = width, pathEffect = effect))
    }
}

private fun blob(c: Offset, r: Float, t: Float, life: Float): Path {
    val path = Path()
    val steps = 96
    for (i in 0..steps) {
        val a = i.toFloat() / steps * 2f * PI.toFloat()
        val w = 1f + (0.03f * sin(a * 3f + t * 0.9f) + 0.02f * sin(a * 5f - t * 1.3f)) * life
        val x = c.x + cos(a) * r * w
        val y = c.y + sin(a) * r * w
        if (i == 0) path.moveTo(x, y) else path.lineTo(x, y)
    }
    path.close()
    return path
}

// ---------- One Halo across screens ----------

/** Lets each screen's Halo be the same shared element, so it glides from screen to screen. */
@OptIn(ExperimentalSharedTransitionApi::class)
class HaloScopes(val shared: SharedTransitionScope, val visibility: AnimatedVisibilityScope)

val LocalHaloScopes = compositionLocalOf<HaloScopes?> { null }

@OptIn(ExperimentalSharedTransitionApi::class)
@Composable
fun Modifier.sharedHalo(): Modifier {
    val scopes = LocalHaloScopes.current ?: return this
    if (LocalReduceMotion.current) return this
    return with(scopes.shared) {
        this@sharedHalo.sharedElement(
            rememberSharedContentState(key = "halo"),
            animatedVisibilityScope = scopes.visibility,
        )
    }
}

/** Where the in-call Halo sits on screen, so the alert can flood outward from it. */
class HaloAnchor {
    var center by mutableStateOf(Offset.Unspecified)
}

val LocalHaloAnchor = staticCompositionLocalOf { HaloAnchor() }

fun Modifier.reportCenter(anchor: HaloAnchor): Modifier =
    onGloballyPositioned { anchor.center = it.boundsInRoot().center }
