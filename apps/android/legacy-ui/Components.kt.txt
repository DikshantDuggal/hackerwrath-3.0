/** A switch whose knob lands with a little give. */
@Composable
fun HaloSwitch(checked: Boolean, modifier: Modifier = Modifier) {
    val knob by animateDpAsState(if (checked) 27.dp else 3.dp, giveSpec(), label = "knob")
    val track by animateColorAsState(if (checked) EditorialBlue else Outline, calmSpec(Motion.SHORT), label = "track")

        Box(Modifier.offset(x = knob, y = 3.dp).size(28.dp).clip(CircleShape).background(PureWhite))
    }
}

/** Three languages in their own scripts; a white pill slides to the chosen one. */
@Composable
fun LanguageSegmented(selected: AppLanguage, onSelect: (AppLanguage) -> Unit, modifier: Modifier = Modifier) {
    val options = listOf(AppLanguage.English to "English", AppLanguage.Hindi to "हिन्दी", AppLanguage.Punjabi to "ਪੰਜਾਬੀ")
    val index = options.indexOfFirst { it.first == selected }.coerceAtLeast(0)
    val slide by animateFloatAsState(index.toFloat(), giveSpec(), label = "segment")
    BoxWithConstraints(
        modifier.fillMaxWidth().clip(RoundedCornerShape(16.dp)).background(Color(0xFFEDEFF2)).padding(4.dp),
    ) {
        val cell = maxWidth / options.size
        Box(
            Modifier
                .offset { IntOffset((cell.toPx() * slide).toInt(), 0) }
                .width(cell)
                .height(48.dp)
                .clip(RoundedCornerShape(12.dp))
                .background(PureWhite)
        )
        Row(Modifier.fillMaxWidth()) {
            options.forEach { (language, label) ->
                Box(
                    contentAlignment = Alignment.Center,
                    modifier = Modifier
                        .weight(1f)
                        .heightIn(min = 48.dp)
                        .clip(RoundedCornerShape(12.dp))
                        .selectable(selected = language == selected, role = Role.RadioButton, onClick = { onSelect(language) }),
                ) { Text(label, style = MaterialTheme.typography.titleMedium) }
            }
        }
    }
}

/** Big round Answer / Decline. Answer sends out a soft repeating ring. */
@Composable
fun RingButton(icon: ImageVector, label: String, color: Color, onClick: () -> Unit, pulse: Boolean = false) {
    val interaction = remember { MutableInteractionSource() }
    val reduceMotion = LocalReduceMotion.current
    val ring = rememberInfiniteTransition(label = "ring")
    val q by ring.animateFloat(0f, 1f, infiniteRepeatable(tween(1600, easing = FastOutSlowInEasing), RepeatMode.Restart), label = "q")
    Column(horizontalAlignment = Alignment.CenterHorizontally, verticalArrangement = Arrangement.spacedBy(10.dp)) {
        Box(
            contentAlignment = Alignment.Center,
            modifier = Modifier
                .size(80.dp)
                .drawBehind {
                    if (pulse && !reduceMotion) {
                        drawCircle(color.copy(alpha = 0.6f * (1f - q)), radius = size.minDimension / 2f * (1f + 0.4f * q), style = Stroke(2.dp.toPx()))
                    }
                }
                .pressScale(interaction)
                .clip(CircleShape)
                .background(color)
                .toggleable(value = false, interactionSource = interaction, indication = null, role = Role.Button, onValueChange = { onClick() })
                .semantics { contentDescription = label },
        ) { Icon(icon, contentDescription = null, tint = Color.White, modifier = Modifier.size(32.dp)) }
        Text(label, style = MaterialTheme.typography.bodyMedium, color = OnNightMuted)
    }
}

/** The thin bar under the caller's number: it fills with the call, and marks appear as Halo notices things. */
@Composable
fun CallRibbon(timeline: List<TimelineEntry>, seconds: Long, modifier: Modifier = Modifier) {
    val span = maxOf(60L, seconds).toFloat()
    val fill by animateFloatAsState(seconds / span, tween(900, easing = LinearEasing), label = "ribbon")
    Canvas(modifier.width(220.dp).height(14.dp).clearAndSetSemantics { }) {
        val y = size.height / 2f
        val h = 6.dp.toPx()
        drawLine(Color.White.copy(alpha = 0.1f), Offset(h / 2, y), Offset(size.width - h / 2, y), strokeWidth = h, cap = StrokeCap.Round)
        if (fill > 0f) {
            drawLine(HaloListening.copy(alpha = 0.55f), Offset(h / 2, y), Offset(h / 2 + (size.width - h) * fill, y), strokeWidth = h, cap = StrokeCap.Round)
        }
        timeline.forEach { e ->
            val x = h / 2 + (size.width - h) * (e.atSeconds / span).coerceIn(0f, 1f)
            drawCircle(Night, radius = 7.dp.toPx(), center = Offset(x, y))
            drawCircle(e.kind.color(), radius = 5.dp.toPx(), center = Offset(x, y))
        }
    }
}

private const val HOLD_MS = 1_500

/**
 * Keep talking after a warning needs a deliberate press-and-hold; a quick tap does nothing.
 * TalkBack offers it as a long-press action ("double-tap and hold").
 */
@Composable
fun HoldToConfirmButton(label: String, holdingLabel: String, onConfirmed: () -> Unit, modifier: Modifier = Modifier) {
    val progress = remember { Animatable(0f) }
    val scope = rememberCoroutineScope()
    val haptics = LocalHapticFeedback.current
    var holding by remember { mutableStateOf(false) }
    Box(
        contentAlignment = Alignment.Center,
        modifier = modifier
            .fillMaxWidth()
            .heightIn(min = 56.dp)
            .clip(RoundedCornerShape(18.dp))
            .border(1.5.dp, Color.White.copy(alpha = 0.55f), RoundedCornerShape(18.dp))
            .semantics {
                role = Role.Button
                contentDescription = label
                onLongClick(label = label) { onConfirmed(); true }
            }
            .pointerHold(
                onStart = {
                    holding = true
                    haptics.performHapticFeedback(HapticFeedbackType.TextHandleMove)
                    scope.launch {
                        val remaining = ((1f - progress.value) * HOLD_MS).toInt()
                        progress.animateTo(1f, tween(remaining, easing = LinearEasing))
                        haptics.performHapticFeedback(HapticFeedbackType.LongPress)
                        holding = false
                        onConfirmed()
                    }
                },
                onRelease = { job ->
                    holding = false
                    if (progress.value < 1f) {
                        job?.cancel()
                        scope.launch { progress.animateTo(0f, tween(250)) }
                    }
                },
            ),
    ) {
        Box(
            Modifier
                .matchParentSizeFill()
                .graphicsLayer {
                    scaleX = progress.value
                    transformOrigin = androidx.compose.ui.graphics.TransformOrigin(0f, 0.5f)
                }
                .background(Color.White.copy(alpha = 0.22f))
        )
        Text(
            if (holding) holdingLabel else label,
            style = MaterialTheme.typography.labelLarge.copy(fontWeight = FontWeight.SemiBold),
            color = Color.White,
            modifier = Modifier.padding(horizontal = 16.dp, vertical = 14.dp),
        )
    }
}

private fun Modifier.matchParentSizeFill(): Modifier = this.fillMaxWidth().fillMaxHeight()

/** Calls [onStart] on press and [onRelease] with the job [onStart] returned when the finger lifts. */
private fun Modifier.pointerHold(onStart: () -> Job, onRelease: (Job?) -> Unit): Modifier =
    this.then(
        Modifier.pointerInputHold(onStart, onRelease)
    )

private fun Modifier.pointerInputHold(onStart: () -> Job, onRelease: (Job?) -> Unit): Modifier =
    androidx.compose.ui.input.pointer.pointerInput(Unit) {
        detectTapGestures(onPress = {
            val job = onStart()
            tryAwaitRelease()
            onRelease(job)
        })
    }.let { this.then(it) }

/**
 * Shows which way the money (or the key to the account) really moves: the caller's story as a
 * faded, crossed-out arrow, and the truth as a solid arrow with a token travelling along it.
 */
@Composable
fun TrickDiagram(type: RequestType, modifier: Modifier = Modifier) {
    val (said, real) = when (type) {
        RequestType.QrReceive -> R.string.trick_qr_said to R.string.trick_qr_real
        RequestType.PinReceive -> R.string.trick_pin_said to R.string.trick_pin_real
        RequestType.OtpShare -> R.string.trick_otp_said to R.string.trick_otp_real
    }
    val tokenIcon = when (type) {
        RequestType.QrReceive -> Icons.Outlined.CurrencyRupee
        RequestType.PinReceive -> Icons.Outlined.Dialpad
        RequestType.OtpShare -> Icons.Outlined.Key
    }
    val reduceMotion = LocalReduceMotion.current
    val travel = rememberInfiniteTransition(label = "trick")
    val moving by travel.animateFloat(0f, 1f, infiniteRepeatable(tween(1800, easing = FastOutSlowInEasing), RepeatMode.Restart), label = "token")
    val realText = stringResource(real)
    Column(
        modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(20.dp))
            .background(Color.Black.copy(alpha = 0.18f))
            .border(1.dp, Color.White.copy(alpha = 0.18f), RoundedCornerShape(20.dp))
            .semantics(mergeDescendants = true) { contentDescription = realText }
            .padding(horizontal = 12.dp, vertical = 12.dp),
        verticalArrangement = Arrangement.spacedBy(4.dp),
    ) {
        Text(
            stringResource(said),
            style = MaterialTheme.typography.labelMedium.copy(fontWeight = FontWeight.Normal),
            color = Color.White.copy(alpha = 0.75f),
            textDecoration = TextDecoration.LineThrough,
            textAlign = TextAlign.Center,
            modifier = Modifier.fillMaxWidth(),
        )
        BoxWithConstraints(Modifier.fillMaxWidth().height(100.dp)) {
            val density = LocalDensity.current
            val w = with(density) { maxWidth.toPx() }
            val h = with(density) { maxHeight.toPx() }
            val nodeR = with(density) { 26.dp.toPx() }
            val geo = remember(w, h, nodeR) { TrickGeometry(w, h, nodeR) }
            Canvas(Modifier.fillMaxWidth().fillMaxHeight()) {
                val dash = PathEffect.dashPathEffect(floatArrayOf(5.dp.toPx(), 6.dp.toPx()))
                drawPath(geo.saidPath, Color.White.copy(alpha = 0.45f), style = Stroke(2.dp.toPx(), pathEffect = dash))
                drawPath(geo.realPath, Color.White, style = Stroke(2.5.dp.toPx(), cap = StrokeCap.Round))
                drawPath(geo.arrowHead, Color.White)
                drawCircle(Color.White.copy(alpha = 0.16f), nodeR, geo.you)
                drawCircle(Color.White, nodeR, geo.you, style = Stroke(1.5.dp.toPx()))
                drawCircle(Color.Black.copy(alpha = 0.2f), nodeR, geo.them)
                drawCircle(Color.White, nodeR, geo.them, style = Stroke(1.5.dp.toPx()))
            }
            Text(
                stringResource(R.string.trick_you),
                style = MaterialTheme.typography.labelMedium.copy(fontWeight = FontWeight.Bold),
                color = Color.White,
                textAlign = TextAlign.Center,
                modifier = Modifier.align(Alignment.CenterStart).width(56.dp),
            )
            Text(
                stringResource(R.string.trick_caller),
                style = MaterialTheme.typography.labelMedium.copy(fontWeight = FontWeight.Bold),
                color = Color.White,
                textAlign = TextAlign.Center,
                modifier = Modifier.align(Alignment.CenterEnd).width(56.dp),
            )
            val tokenSize = 28.dp
            val half = with(density) { (tokenSize / 2).toPx() }
            Box(
                contentAlignment = Alignment.Center,
                modifier = Modifier
                    .offset {
                        val p = geo.realAt(if (reduceMotion) 0.5f else moving)
                        IntOffset((p.x - half).toInt(), (p.y - half).toInt())
                    }
                    .size(tokenSize)
                    .clip(CircleShape)
                    .background(TokenGold)
                    .border(2.dp, Color.White, CircleShape),
            ) { Icon(tokenIcon, contentDescription = null, tint = OnTokenGold, modifier = Modifier.size(16.dp)) }
        }
        Text(
            realText,
            style = MaterialTheme.typography.labelMedium.copy(fontWeight = FontWeight.Bold),
            color = Color.White,
            textAlign = TextAlign.Center,
            modifier = Modifier.fillMaxWidth(),
        )
    }
}

private class TrickGeometry(w: Float, h: Float, nodeR: Float) {
    val you = Offset(nodeR + 4f, h / 2f)
    val them = Offset(w - nodeR - 4f, h / 2f)
    private val saidFrom = Offset(them.x - nodeR * 0.7f, h / 2f - nodeR * 0.75f)
    private val saidCtrl = Offset(w / 2f, -h * 0.05f)
    private val saidTo = Offset(you.x + nodeR * 0.7f, h / 2f - nodeR * 0.75f)
    private val realFrom = Offset(you.x + nodeR * 0.7f, h / 2f + nodeR * 0.75f)
    private val realCtrl = Offset(w / 2f, h * 1.05f)
    private val realTo = Offset(them.x - nodeR * 0.7f, h / 2f + nodeR * 0.75f)

    val saidPath = Path().apply { moveTo(saidFrom.x, saidFrom.y); quadraticTo(saidCtrl.x, saidCtrl.y, saidTo.x, saidTo.y) }
    val realPath = Path().apply { moveTo(realFrom.x, realFrom.y); quadraticTo(realCtrl.x, realCtrl.y, realTo.x, realTo.y) }

    /** A small filled triangle at the end of the real arrow, pointing along it. */
    val arrowHead: Path = run {
        val dir = (realTo - realCtrl).let { d -> d / d.getDistance() }
        val normal = Offset(-dir.y, dir.x)
        val size = 9f * (nodeR / 26f)
        val tip = realTo + dir * size * 0.6f
        val back = realTo - dir * size
        Path().apply {
            moveTo(tip.x, tip.y)
            lineTo(back.x + normal.x * size * 0.6f, back.y + normal.y * size * 0.6f)
            lineTo(back.x - normal.x * size * 0.6f, back.y - normal.y * size * 0.6f)
            close()
        }
    }

    fun realAt(t: Float): Offset {
        val u = 1f - t
        return realFrom * (u * u) + realCtrl * (2f * u * t) + realTo * (t * t)
    }
}

/** Four bars that move while the spoken warning plays. */
@Composable
fun SpokenBars(modifier: Modifier = Modifier, color: Color = Color.White) {
    val reduceMotion = LocalReduceMotion.current
    val eq = rememberInfiniteTransition(label = "eq")
    val phase by eq.animateFloat(0f, 1f, infiniteRepeatable(tween(900, easing = LinearEasing), RepeatMode.Restart), label = "eqPhase")
    Canvas(modifier.size(width = 24.dp, height = 18.dp).clearAndSetSemantics { }) {
        val gap = size.width / 4f
        for (i in 0 until 4) {
            val local = ((phase + i * 0.15f) % 1f)
            val k = if (reduceMotion) 0.5f else 0.5f - 0.5f * kotlin.math.cos(local * 2f * Math.PI.toFloat())
            val bh = size.height * (0.33f + 0.67f * k)
            val x = gap * i + gap / 2f
            drawLine(color, Offset(x, (size.height - bh) / 2f), Offset(x, (size.height + bh) / 2f), strokeWidth = 3.dp.toPx(), cap = StrokeCap.Round)
        }
    }
}

// ---------- Recap ----------

@Composable
fun timelineLabel(entry: TimelineEntry): String {
    val request = when (entry.type) {
        RequestType.QrReceive -> stringResource(R.string.req_qr)
        RequestType.PinReceive -> stringResource(R.string.req_pin)
        RequestType.OtpShare -> stringResource(R.string.req_otp)
        null -> null
    }
    return when (entry.kind) {
        TimelineKind.Checking -> stringResource(R.string.tl_checking)
        TimelineKind.Possible -> stringResource(R.string.tl_level, stringResource(R.string.level_possible), request.orEmpty())
        TimelineKind.High -> stringResource(R.string.tl_level, stringResource(R.string.level_high), request.orEmpty())
        TimelineKind.Paused -> stringResource(R.string.tl_paused)
        TimelineKind.Resumed -> stringResource(R.string.tl_resumed)
        TimelineKind.Continued -> stringResource(R.string.tl_continued)
        TimelineKind.YouEnded -> stringResource(R.string.tl_you_ended)
        TimelineKind.CallerEnded -> stringResource(R.string.tl_caller_ended)
    }
}

fun clock(seconds: Long): String = "%d:%02d".format(seconds / 60, seconds % 60)

/** The call as a bar with coloured dots, then each moment as a line. */
@Composable
fun TimelineCard(timeline: List<TimelineEntry>, totalSeconds: Long, modifier: Modifier = Modifier) {
    val total = maxOf(1L, totalSeconds).toFloat()
    val drawn = remember { Animatable(0f) }
    val reduceMotion = LocalReduceMotion.current
    androidx.compose.runtime.LaunchedEffect(Unit) {
        if (reduceMotion) drawn.snapTo(1f) else drawn.animateTo(1f, tween(900, delayMillis = 250, easing = Motion.EaseOut))
    }
    Column(
        modifier.fillMaxWidth().clip(RoundedCornerShape(20.dp)).background(PureWhite).border(1.dp, CoolGrey, RoundedCornerShape(20.dp)).padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(10.dp),
    ) {
        SectionLabel(stringResource(R.string.recap_label))
        Canvas(Modifier.fillMaxWidth().height(20.dp).clearAndSetSemantics { }) {
            val y = size.height / 2f
            val h = 8.dp.toPx()
            drawLine(Color(0xFFEDEFF2), Offset(h / 2, y), Offset(size.width - h / 2, y), strokeWidth = h, cap = StrokeCap.Round)
            drawLine(EditorialBlueContainer, Offset(h / 2, y), Offset(h / 2 + (size.width - h) * drawn.value, y), strokeWidth = h, cap = StrokeCap.Round)
            timeline.forEach { e ->
                val f = e.atSeconds / total
                if (f <= drawn.value) {
                    val x = h / 2 + (size.width - h) * f.coerceIn(0f, 1f)
                    drawCircle(PureWhite, radius = 9.dp.toPx(), center = Offset(x, y))
                    drawCircle(e.kind.color(), radius = 6.dp.toPx(), center = Offset(x, y))
                }
            }
        }
        Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween) {
            Text("0:00", style = MaterialTheme.typography.labelMedium.copy(fontWeight = FontWeight.Normal), color = Slate)
            Text(clock(totalSeconds), style = MaterialTheme.typography.labelMedium.copy(fontWeight = FontWeight.Normal), color = Slate)
        }
        timeline.forEach { e ->
            Row(verticalAlignment = Alignment.CenterVertically, horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                Text(clock(e.atSeconds), style = MaterialTheme.typography.labelMedium.copy(fontWeight = FontWeight.Normal), color = Slate, modifier = Modifier.width(44.dp))
                Box(Modifier.size(10.dp).clip(CircleShape).background(e.kind.color()))
                Text(timelineLabel(e), style = MaterialTheme.typography.bodyMedium, modifier = Modifier.weight(1f))
            }
        }
    }
}
